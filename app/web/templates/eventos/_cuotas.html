{# Cuotas de un evento (Local, Empate, Visitante), en columnas fijas para que se alineen entre partidos. Cada cuota activa de un evento que acepta apuestas es un botón HTMX que la agrega al boleto (panel de Daniel, rutas /web/boleto/*); sin sesión enlaza a /login. Parámetros: evento, usuario (o none), apostable (PROGRAMADO y aún no inicia), con_empate: true/false para forzar la columna de empate; none la decide según las cuotas del evento, grande: variante de la pantalla de detalle. #} {% macro cuotas(evento, usuario, apostable, con_empate=none, grande=false) %} {% set activas = evento.cuotas | selectattr("activa") | list %} {% set tiene_empate = con_empate if con_empate is not none else (activas | selectattr("seleccion.value", "equalto", "EMPATE") | list | length > 0) %} {% set minimo = (activas | map(attribute="valor") | min) if activas | length > 1 else none %} {% set columnas = [("LOCAL", "Local")] + ([("EMPATE", "Empate")] if tiene_empate else []) + [("VISITANTE", "Visitante")] %}
{% for clave, etiqueta in columnas %} {% set cuota = activas | selectattr("seleccion.value", "equalto", clave) | first %} {% if cuota %} {% set favorita = ' cuota--favorita' if minimo is not none and cuota.valor == minimo else '' %} {% if apostable and usuario %} {% elif apostable %} {{ etiqueta }} {{ cuota.valor }} Inicia sesión para apostar {% else %}
{{ etiqueta }} {{ cuota.valor }}
{% endif %} {% else %} – {% endif %} {% endfor %}
{% endmacro %}
